cmake_minimum_required(VERSION 3.8)
project(soarm4_moveit_config)
find_package(ament_cmake REQUIRED)
install(DIRECTORY config launch audit_tools audit_model runtime_deps DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/planning_state.py scripts/plan_joint_goal.py scripts/fine_collision_service.py scripts/home_manager.py scripts/go_home.py scripts/pick_place_manager.py scripts/pick_place.py scripts/pick_place_core.py scripts/plan_cache.py DESTINATION lib/${PROJECT_NAME})
find_package(moveit_core REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
add_library(soarm4_preview_controller_manager SHARED src/preview_controller_manager.cpp)
ament_target_dependencies(soarm4_preview_controller_manager moveit_core pluginlib rclcpp)
pluginlib_export_plugin_description_file(moveit_core preview_controller_plugin.xml)
install(TARGETS soarm4_preview_controller_manager LIBRARY DESTINATION lib)
find_package(moveit_msgs REQUIRED)
add_library(soarm4_fine_collision_adapter SHARED src/fine_collision_adapter.cpp)
ament_target_dependencies(soarm4_fine_collision_adapter moveit_core pluginlib rclcpp moveit_msgs)
pluginlib_export_plugin_description_file(moveit_core fine_collision_plugin.xml)
install(TARGETS soarm4_fine_collision_adapter LIBRARY DESTINATION lib)
find_package(rviz_common REQUIRED)
find_package(std_srvs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
add_library(soarm4_home_panel SHARED src/home_panel.cpp)
ament_target_dependencies(soarm4_home_panel rviz_common rclcpp std_srvs std_msgs pluginlib)
target_link_libraries(soarm4_home_panel Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common home_panel_plugin.xml)
install(TARGETS soarm4_home_panel LIBRARY DESTINATION lib)
ament_package()

install(FILES PICK_PLACE_KO.md DESTINATION share/${PROJECT_NAME})
